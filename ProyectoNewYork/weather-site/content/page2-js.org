<script>
  // Extraemos la conversión exacta a 32 bits de la memoria de JS
  function floatToIEEE754(numero) {
    let buffer = new ArrayBuffer(4);
    let vistaFloat = new Float32Array(buffer);
    let vistaInt = new Uint32Array(buffer);
    
    vistaFloat[0] = numero;
    let bits = vistaInt[0].toString(2).padStart(32, '0');
    
    return {
      signo: bits[0],
      exponente: bits.substring(1, 9),
      mantisa: bits.substring(9, 32)
    };
  }

  // Generamos una explicación simplificada del proceso para el HTML
  function generarExplicacion(num1, num2, resultado, conversion) {
      let signoFinal = (num1 * num2 < 0) ? "1 (Negativo)" : "0 (Positivo)";
      let expEntero = parseInt(conversion.exponente, 2);
      let expReal = expEntero - 127;

      let explicacion = `
<h4>Procedimiento Matemático:</h4>
<pre>
1. Análisis de Signos:
   Signo = Signo(N1) XOR Signo(N2)
   El resultado es ${signoFinal}. Por lo tanto, el bit de signo es: ${conversion.signo}.

2. Exponente (Basado en el resultado de ${resultado}):
   Exponente Real = ${expReal}
   Exponente Sesgado (Suma de 127) = ${expReal} + 127 = ${expEntero}
   Convertido a Binario (8 bits) = ${conversion.exponente}

3. Mantisa (Fracción normalizada):
   Se obtiene al quitar el '1.' implícito de la notación científica binaria.
   Rellenando con ceros hasta 23 bits: ${conversion.mantisa}

4. Ensamblaje IEEE 754:
   [Signo] [Exponente] [Mantisa]
</pre>`;
      return explicacion;
  }

  function procesarMultiplicacion() {
    let val1 = parseFloat(document.getElementById('n1').value);
    let val2 = parseFloat(document.getElementById('n2').value);
    
    if (isNaN(val1) || isNaN(val2)) {
      alert("Por favor, ingresa dos números válidos.");
      return;
    }

    let resultado = val1 * val2;
    let conversion = floatToIEEE754(resultado);

    document.getElementById('out-op').innerText = `${val1} × ${val2}`;
    document.getElementById('out-dec').innerText = resultado;
    
    // Inyectamos el paso a paso
    document.getElementById('out-pasos').innerHTML = generarExplicacion(val1, val2, resultado, conversion);

    // Inyectamos los colores finales
    document.getElementById('out-signo').innerText = conversion.signo;
    document.getElementById('out-exp').innerText = conversion.exponente;
    document.getElementById('out-mantisa').innerText = conversion.mantisa;
    
    document.getElementById('panel-resultado').style.display = 'block';
  }
</script>
